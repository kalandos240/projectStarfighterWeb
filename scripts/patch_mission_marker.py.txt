from pathlib import Path

path = Path('upstream/src/Starfighter.c')
source = path.read_text()
old = '''#ifdef __EMSCRIPTEN__
\t\t\t\tEM_ASM({ if (globalThis.starfighterGameplayStart) globalThis.starfighterGameplayStart(); });
#endif
\t\t\t\tsection = game_mainLoop();
'''
new = '''#ifdef __EMSCRIPTEN__
\t\t\t\tEM_ASM({
\t\t\t\t\tdocument.documentElement.setAttribute('data-starfighter-mission-started', '1');
\t\t\t\t\tif (globalThis.starfighterGameplayStart) globalThis.starfighterGameplayStart();
\t\t\t\t});
#endif
\t\t\t\tsection = game_mainLoop();
'''
if new not in source:
    if old not in source:
        raise SystemExit('mission gameplay marker not found')
    source = source.replace(old, new, 1)
path.write_text(source)

from pathlib import Path

path = Path('upstream/src/title.c')
source = path.read_text()
old = '''\t\t\tif ((now - then >= 33250) || (skip))
\t\t\t{
\t\t\t\toptionRec.x = screen->w / 2 - optionRec.w / 2;
'''
new = '''\t\t\tif ((now - then >= 33250) || (skip))
\t\t\t{
#ifdef __EMSCRIPTEN__
\t\t\t\tEM_ASM({ document.documentElement.setAttribute('data-starfighter-menu-ready', '1'); });
#endif
\t\t\t\toptionRec.x = screen->w / 2 - optionRec.w / 2;
'''
if new not in source:
    if old not in source:
        raise SystemExit('title menu marker not found')
    source = source.replace(old, new, 1)
path.write_text(source)

# The text field is a fixed array, so its address is always non-null. Keep only
# the image check the original shortcut actually needs.
gfx_path = Path('upstream/src/gfx.c')
gfx_source = gfx_path.read_text()
warning_old = '''\tif (gfx_textSprites[index].text && gfx_textSprites[index].image
\t\t\t&& (gfx_textSprites[index].fontColor == fontColor)
'''
warning_new = '''\tif (gfx_textSprites[index].image
\t\t\t&& (gfx_textSprites[index].fontColor == fontColor)
'''
if warning_new not in gfx_source:
    if warning_old not in gfx_source:
        raise SystemExit('text object shortcut marker not found')
    gfx_source = gfx_source.replace(warning_old, warning_new, 1)
gfx_path.write_text(gfx_source)

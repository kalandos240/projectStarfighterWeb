from pathlib import Path
import shutil

path = Path('upstream/src/title.c')
source = path.read_text()
old = '''\t\t\tif ((now - then >= 33250) || (skip))
\t\t\t{
\t\t\t\toptionRec.x = screen->w / 2 - optionRec.w / 2;
'''
new = '''\t\t\tif ((now - then >= 33250) || (skip))
\t\t\t{
#ifdef __EMSCRIPTEN__
\t\t\t\tEM_ASM({ document.documentElement.setAttribute('data-starfighter-menu-ready', '1'); });
#endif
\t\t\t\toptionRec.x = screen->w / 2 - optionRec.w / 2;
'''
if new not in source:
    if old not in source:
        raise SystemExit('title menu marker not found')
    source = source.replace(old, new, 1)

difficulty_old = '''\t\t\t\t\t\t\tmenuType = MENU_DIFFICULTY;
\t\t\t\t\t\t\tselectedOption = 1;
'''
difficulty_new = '''\t\t\t\t\t\t\tmenuType = MENU_DIFFICULTY;
\t\t\t\t\t\t\tselectedOption = 1;
#ifdef __EMSCRIPTEN__
\t\t\t\t\t\t\tEM_ASM({ document.documentElement.setAttribute('data-starfighter-difficulty-ready', '1'); });
#endif
'''
if difficulty_new not in source:
    if difficulty_old not in source:
        raise SystemExit('difficulty menu marker not found')
    source = source.replace(difficulty_old, difficulty_new, 1)
path.write_text(source)

gfx_path = Path('upstream/src/gfx.c')
gfx_source = gfx_path.read_text()
warning_old = '''\tif (gfx_textSprites[index].text && gfx_textSprites[index].image
\t\t\t&& (gfx_textSprites[index].fontColor == fontColor)
'''
warning_new = '''\tif (gfx_textSprites[index].image
\t\t\t&& (gfx_textSprites[index].fontColor == fontColor)
'''
if warning_new not in gfx_source:
    if warning_old not in gfx_source:
        raise SystemExit('text object shortcut marker not found')
    gfx_source = gfx_source.replace(warning_old, warning_new, 1)
gfx_path.write_text(gfx_source)

shutil.copyfile('web/web_i18n.c.txt', 'upstream/src/web_i18n.c')
i18n_path = Path('upstream/src/web_i18n.c')
i18n_source = i18n_path.read_text()
i18n_probe_old = "    const lang = String(sdkLang || navigator.language || '').toLowerCase();\n"
i18n_probe_new = "    const lang = String(sdkLang || navigator.language || String()).toLowerCase();\n"
if i18n_probe_new not in i18n_source:
    if i18n_probe_old not in i18n_source:
        raise SystemExit('web language probe marker not found')
    i18n_source = i18n_source.replace(i18n_probe_old, i18n_probe_new, 1)
i18n_marker = '    {NULL, NULL}\n};'
i18n_fragments = []
for fragment_path in sorted(Path('web').glob('i18n-*.inc')):
    fragment = fragment_path.read_text()
    if fragment.strip():
        i18n_fragments.append(fragment.rstrip() + '\n')
if i18n_fragments:
    if i18n_marker not in i18n_source:
        raise SystemExit('web translation table terminator not found')
    i18n_source = i18n_source.replace(
        i18n_marker,
        '\n' + ''.join(i18n_fragments) + '\n' + i18n_marker,
        1,
    )
i18n_path.write_text(i18n_source)

defs_path = Path('upstream/src/defs.h')
defs_source = defs_path.read_text()
i18n_old = '#define _(s) gettext(s)\n'
i18n_new = '''#ifdef __EMSCRIPTEN__
const char *starfighter_web_gettext(const char *source);
#define _(s) starfighter_web_gettext(s)
#else
#define _(s) gettext(s)
#endif
'''
if i18n_new not in defs_source:
    if i18n_old not in defs_source:
        raise SystemExit('gettext macro marker not found')
    defs_source = defs_source.replace(i18n_old, i18n_new, 1)
defs_path.write_text(defs_source)

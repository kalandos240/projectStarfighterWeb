from pathlib import Path


def add_emscripten_include(path):
    source = path.read_text()
    old = '#include "SDL.h"\n'
    new = '#include "SDL.h"\n\n#ifdef __EMSCRIPTEN__\n#include <emscripten.h>\n#endif\n'
    if new not in source:
        if old not in source:
            raise SystemExit(f'SDL include marker not found in {path}')
        source = source.replace(old, new, 1)
    path.write_text(source)


# Platform pause/resume hook in the central frame delay.
game_path = Path('upstream/src/game.c')
add_emscripten_include(game_path)
source = game_path.read_text()
old = '''void game_delayFrame()
{
\tUint32 now = SDL_GetTicks();

\t// Add 16 2/3 (= 1000 / 60) to frameLimit
\tframeLimit += 16;
\tthirds += 2;
\twhile (thirds >= 3)
\t{
\t\tthirds -= 3;
\t\tframeLimit++;
\t}

\tif(now < frameLimit)
\t\tSDL_Delay(frameLimit - now);
\telse
\t\tframeLimit = now;
}
'''
new = '''void game_delayFrame()
{
#ifdef __EMSCRIPTEN__
\tif (EM_ASM_INT({ return globalThis.starfighterPlatformPaused ? 1 : 0; }))
\t{
#ifndef NOSOUND
\t\tint resumeMusic = Mix_PlayingMusic() && !Mix_PausedMusic();
\t\tif (resumeMusic)
\t\t\tMix_PauseMusic();
\t\tMix_Pause(-1);
#endif

\t\twhile (EM_ASM_INT({ return globalThis.starfighterPlatformPaused ? 1 : 0; }))
\t\t\tSDL_Delay(50);

#ifndef NOSOUND
\t\tif (resumeMusic)
\t\t\tMix_ResumeMusic();
\t\tMix_Resume(-1);
#endif
\t\tframeLimit = SDL_GetTicks();
\t}
#endif

\tUint32 now = SDL_GetTicks();

\t// Add 16 2/3 (= 1000 / 60) to frameLimit
\tframeLimit += 16;
\tthirds += 2;
\twhile (thirds >= 3)
\t{
\t\tthirds -= 3;
\t\tframeLimit++;
\t}

\tif(now < frameLimit)
\t\tSDL_Delay(frameLimit - now);
\telse
\t\tframeLimit = now;
}
'''
if new not in source:
    if old not in source:
        raise SystemExit('game_delayFrame marker not found')
    source = source.replace(old, new, 1)
game_path.write_text(source)

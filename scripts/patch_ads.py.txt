from pathlib import Path

path = Path('upstream/src/Starfighter.c')
source = path.read_text()
old = '''#ifdef __EMSCRIPTEN__
\t\t\t\tEM_ASM({ if (globalThis.starfighterGameplayStop) globalThis.starfighterGameplayStop(); });
#endif
\t\t\t\tbreak;
'''
new = '''#ifdef __EMSCRIPTEN__
\t\t\t\tEM_ASM({
\t\t\t\t\tif (globalThis.starfighterGameplayStop) globalThis.starfighterGameplayStop();
\t\t\t\t\tif (globalThis.starfighterMissionEnded) globalThis.starfighterMissionEnded();
\t\t\t\t});
#endif
\t\t\t\tbreak;
'''
if new not in source:
    if old not in source:
        raise SystemExit('mission end marker not found')
    source = source.replace(old, new, 1)
path.write_text(source)

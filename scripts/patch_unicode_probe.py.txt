from pathlib import Path

path = Path('upstream/src/title.c')
source = path.read_text()
old = '''\tgfx_loadSprites();

\tscreen_clear(black);
'''
new = '''\tgfx_loadSprites();

#ifdef __EMSCRIPTEN__
\tSDL_Surface *webUnicodeTest = gfx_createSurface(256, 32);
\tgfx_renderUnicode("Проверка", 0, 0, FONT_WHITE, 0, webUnicodeTest);
\tSDL_FreeSurface(webUnicodeTest);
\tEM_ASM({ document.documentElement.setAttribute('data-starfighter-unicode-test', '1'); });
#endif

\tscreen_clear(black);
'''
if new not in source:
    if old not in source:
        raise SystemExit('unicode probe marker not found')
    source = source.replace(old, new, 1)
path.write_text(source)

exec(Path('scripts/patch_web_menu.py.txt').read_text(), {})
exec(Path('scripts/patch_i18n_campaign.py.txt').read_text(), {})

i18n_path = Path('upstream/src/web_i18n.c')
i18n_source = i18n_path.read_text()
i18n_marker = '    {NULL, NULL}\n};\n'
if i18n_marker not in i18n_source:
    raise SystemExit('web translation sentinel not found for extra localization')
extra_parts = [p.read_text() for p in sorted(Path('web').glob('i18n-*.inc'))]
if extra_parts:
    i18n_source = i18n_source.replace(i18n_marker, '\n'.join(extra_parts) + '\n' + i18n_marker, 1)
i18n_path.write_text(i18n_source)

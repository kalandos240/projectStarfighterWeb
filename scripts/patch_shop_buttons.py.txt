from pathlib import Path

path = Path('upstream/src/shop.c')
text = path.read_text(encoding='utf-8')

needle = "\tshopSelectedItem = SHOP_NOTHING;\n\n\tdrawShop();\n}\n"
replacement = """\tshopSelectedItem = SHOP_NOTHING;\n\n#ifdef __EMSCRIPTEN__\n\t/* Replace English bitmap Buy/Sell sprites with localized web buttons. */\n\tif (gfx_sprites[SP_BUY] != NULL)\n\t\tSDL_FreeSurface(gfx_sprites[SP_BUY]);\n\tif (gfx_sprites[SP_SELL] != NULL)\n\t\tSDL_FreeSurface(gfx_sprites[SP_SELL]);\n\n\tint buyWidth = gfx_unicodeWidth(_(\"Buy\")) + 12;\n\tint sellWidth = gfx_unicodeWidth(_(\"Sell\")) + 12;\n\tif (buyWidth < 40) buyWidth = 40;\n\tif (sellWidth < 40) sellWidth = 40;\n\n\tgfx_sprites[SP_BUY] = gfx_createSurface(buyWidth, 20);\n\tgfx_drawRect(gfx_sprites[SP_BUY], 0, 0, buyWidth - 1, 19, 0xff, 0xff, 0xff);\n\tgfx_drawRect(gfx_sprites[SP_BUY], 1, 1, buyWidth - 3, 17, 0x00, 0x00, 0x00);\n\tgfx_renderUnicode(_(\"Buy\"), 6, 2, FONT_WHITE, 0, gfx_sprites[SP_BUY]);\n\n\tgfx_sprites[SP_SELL] = gfx_createSurface(sellWidth, 20);\n\tgfx_drawRect(gfx_sprites[SP_SELL], 0, 0, sellWidth - 1, 19, 0xff, 0xff, 0xff);\n\tgfx_drawRect(gfx_sprites[SP_SELL], 1, 1, sellWidth - 3, 17, 0x00, 0x00, 0x00);\n\tgfx_renderUnicode(_(\"Sell\"), 6, 2, FONT_WHITE, 0, gfx_sprites[SP_SELL]);\n#endif\n\n\tdrawShop();\n}\n"""

if needle not in text:
    if 'Replace English bitmap Buy/Sell sprites with localized web buttons.' in text:
        raise SystemExit(0)
    raise SystemExit('shop_init() insertion point not found')

text = text.replace(needle, replacement, 1)
text = text.replace(
    'buy_x, buy_y, 24, 16))',
    'buy_x, buy_y, gfx_sprites[SP_BUY]->w, gfx_sprites[SP_BUY]->h))',
    1,
)
text = text.replace(
    'sell_x, sell_y, 24, 16))',
    'sell_x, sell_y, gfx_sprites[SP_SELL]->w, gfx_sprites[SP_SELL]->h))',
    1,
)

path.write_text(text, encoding='utf-8')

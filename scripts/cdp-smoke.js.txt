const fs = require('fs');

const endpoint = 'http://127.0.0.1:9222/json/list';
const targetUrl = 'http://127.0.0.1:8765/index.html';
const deadline = Date.now() + 60000;

function sleep(ms) {
  return new Promise(resolve => setTimeout(resolve, ms));
}

async function findTarget() {
  while (Date.now() < deadline) {
    try {
      const response = await fetch(endpoint);
      const targets = await response.json();
      const target = targets.find(item => item.type === 'page' && item.url.startsWith(targetUrl));
      if (target && target.webSocketDebuggerUrl) return target;
    } catch (_) {}
    await sleep(200);
  }
  throw new Error('Timed out waiting for Chromium DevTools target');
}

async function connectCdp(url) {
  return new Promise((resolve, reject) => {
    const ws = new WebSocket(url);
    ws.onopen = () => resolve(ws);
    ws.onerror = event => reject(new Error('CDP WebSocket connection failed: ' + event.type));
  });
}

async function main() {
  const target = await findTarget();
  const ws = await connectCdp(target.webSocketDebuggerUrl);
  let nextId = 1;
  const pending = new Map();

  ws.onmessage = event => {
    const message = JSON.parse(event.data);
    if (message.id && pending.has(message.id)) {
      const entry = pending.get(message.id);
      pending.delete(message.id);
      if (message.error) entry.reject(new Error(JSON.stringify(message.error)));
      else entry.resolve(message.result || {});
      return;
    }

    if (message.method === 'Runtime.consoleAPICalled') {
      const args = (message.params.args || []).map(arg => arg.value ?? arg.description ?? '').join(' ');
      console.log('[browser console]', args);
    }
  };

  function send(method, params = {}) {
    return new Promise((resolve, reject) => {
      const id = nextId++;
      pending.set(id, { resolve, reject });
      ws.send(JSON.stringify({ id, method, params }));
    });
  }

  async function evaluate(expression) {
    const result = await send('Runtime.evaluate', {
      expression,
      returnByValue: true
    });
    return result.result?.value;
  }

  async function evaluateAwait(expression) {
    const result = await send('Runtime.evaluate', {
      expression,
      returnByValue: true,
      awaitPromise: true
    });
    if (result.exceptionDetails) {
      throw new Error('Browser evaluation failed: ' + JSON.stringify(result.exceptionDetails));
    }
    return result.result?.value;
  }

  async function waitFor(expression, label, timeoutMs = 30000) {
    const stopAt = Date.now() + timeoutMs;
    while (Date.now() < stopAt) {
      try {
        if (await evaluate(expression)) return;
      } catch (_) {}
      await sleep(200);
    }
    throw new Error('Timed out waiting for ' + label);
  }

  async function pressSpace() {
    await send('Input.dispatchKeyEvent', {
      type: 'keyDown',
      key: ' ',
      code: 'Space',
      windowsVirtualKeyCode: 32,
      nativeVirtualKeyCode: 32,
      text: ' '
    });
    await sleep(100);
    await send('Input.dispatchKeyEvent', {
      type: 'keyUp',
      key: ' ',
      code: 'Space',
      windowsVirtualKeyCode: 32,
      nativeVirtualKeyCode: 32
    });
    await sleep(250);
  }

  await send('Runtime.enable');
  await send('Page.enable');

  await waitFor(`(() => {
    const e = document.documentElement;
    return e &&
      e.getAttribute('data-starfighter-runtime-initialized') === '1' &&
      e.getAttribute('data-starfighter-title-ready') === '1' &&
      e.getAttribute('data-starfighter-first-frame') === '1';
  })()`, 'Starfighter first rendered frame', 45000);

  console.log('Core runtime reached first rendered frame');

  await pressSpace();
  await waitFor(`document.documentElement && document.documentElement.getAttribute('data-starfighter-menu-ready') === '1'`, 'Starfighter main menu after Space input', 10000);
  console.log('Keyboard input reached the Starfighter main menu');

  await pressSpace();
  await waitFor(`document.documentElement && document.documentElement.getAttribute('data-starfighter-difficulty-ready') === '1'`, 'Starfighter difficulty menu', 10000);
  console.log('New game selection reached the difficulty menu');

  await pressSpace();
  await waitFor(`globalThis.starfighterGameplayDesired === true`, 'Starfighter mission gameplay start', 30000);
  await sleep(1500);
  console.log('Difficulty confirmation entered actual mission gameplay');

  const markers = await evaluate(`(() => {
    const e = document.documentElement;
    return {
      prerun: e.getAttribute('data-starfighter-prerun'),
      idbfsReady: e.getAttribute('data-starfighter-idbfs-ready'),
      idbfsTimeout: e.getAttribute('data-starfighter-idbfs-timeout'),
      runtime: e.getAttribute('data-starfighter-runtime-initialized'),
      title: e.getAttribute('data-starfighter-title-ready'),
      firstFrame: e.getAttribute('data-starfighter-first-frame'),
      menu: e.getAttribute('data-starfighter-menu-ready'),
      difficulty: e.getAttribute('data-starfighter-difficulty-ready'),
      platformReady: e.getAttribute('data-starfighter-platform-ready'),
      gameplayDesired: globalThis.starfighterGameplayDesired === true
    };
  })()`);
  console.log('Runtime markers:', JSON.stringify(markers));

  const dom = await evaluate('document.documentElement.outerHTML');
  fs.writeFileSync('dist/runtime-dom.html', dom || '');

  const screenshot = await send('Page.captureScreenshot', {
    format: 'png',
    captureBeyondViewport: false
  });
  fs.writeFileSync('dist/runtime-smoke.png', Buffer.from(screenshot.data, 'base64'));

  const persistencePath = '/home/web_user/.config/starfighter/.qa-persistence';
  const persistenceValue = 'starfighter-idbfs-ok';
  await evaluateAwait(`new Promise((resolve, reject) => {
    try {
      FS.writeFile(${JSON.stringify(persistencePath)}, ${JSON.stringify(persistenceValue)});
      FS.syncfs(false, err => err ? reject(String(err)) : resolve(true));
    } catch (err) {
      reject(String(err));
    }
  })`);
  console.log('IDBFS QA marker persisted before reload');

  await send('Page.reload', { ignoreCache: true });
  await sleep(750);
  await waitFor(`(() => {
    const e = document.documentElement;
    return e &&
      e.getAttribute('data-starfighter-runtime-initialized') === '1' &&
      e.getAttribute('data-starfighter-idbfs-ready') === '1';
  })()`, 'Starfighter runtime and IDBFS after reload', 45000);

  await waitFor(`(() => {
    try {
      return FS.readFile(${JSON.stringify(persistencePath)}, { encoding: 'utf8' }) === ${JSON.stringify(persistenceValue)};
    } catch (_) {
      return false;
    }
  })()`, 'persisted IDBFS marker after reload', 10000);
  console.log('IDBFS persistence survived page reload');

  ws.close();
}

main().catch(error => {
  console.error(error.stack || error);
  process.exit(1);
});

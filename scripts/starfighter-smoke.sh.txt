set -euo pipefail

if [ -f emsdk/emsdk_env.sh ]; then
  source emsdk/emsdk_env.sh >/dev/null
fi
node --version

BROWSER="$(command -v google-chrome || command -v chromium || command -v chromium-browser || true)"
if [ -z "$BROWSER" ]; then
  echo "No Chromium-compatible browser found on runner" >&2
  exit 1
fi

cat > dist/sdk.js <<'JS'
globalThis.__qaSdkCalls = [];
globalThis.YaGames = {
  init: async function () {
    globalThis.__qaSdkCalls.push('init');
    return {
      environment: { i18n: { lang: 'ru' } },
      features: {
        LoadingAPI: {
          ready() {
            globalThis.__qaSdkCalls.push('loading-ready');
            document.documentElement.setAttribute('data-qa-loading-ready', '1');
          }
        },
        GameplayAPI: {
          start() { globalThis.__qaSdkCalls.push('gameplay-start'); },
          stop() { globalThis.__qaSdkCalls.push('gameplay-stop'); }
        }
      },
      on() {},
      getPlayer: async function () {
        return {
          getData: async function () { return {}; },
          setData: async function () {}
        };
      },
      adv: { showFullscreenAdv() {} }
    };
  }
};
JS

python3 -m http.server 8765 --directory dist >dist/runtime-server.log 2>&1 &
SERVER_PID=$!
TARGET_URL="http://127.0.0.1:8765/index.html"
echo "Testing Yandex-mocked launch: $TARGET_URL"

PROFILE_DIR="$(mktemp -d)"
mkdir -p "$PROFILE_DIR/Default"
printf '%s\n' '{"intl":{"accept_languages":"en-US,en"}}' > "$PROFILE_DIR/Default/Preferences"

"$BROWSER" \
  --headless=new \
  --no-sandbox \
  --disable-dev-shm-usage \
  --use-angle=swiftshader \
  --enable-unsafe-swiftshader \
  --autoplay-policy=no-user-gesture-required \
  --enable-logging=stderr \
  --v=0 \
  --lang=en-US \
  --window-size=1280,720 \
  --remote-debugging-address=127.0.0.1 \
  --remote-debugging-port=9222 \
  --user-data-dir="$PROFILE_DIR" \
  "$TARGET_URL" \
  >dist/runtime-browser.stdout 2>dist/runtime-browser.log &
BROWSER_PID=$!

cleanup() {
  kill "$BROWSER_PID" 2>/dev/null || true
  kill "$SERVER_PID" 2>/dev/null || true
  sleep 0.2
  rm -f dist/sdk.js
  rm -rf "$PROFILE_DIR" 2>/dev/null || true
}
trap cleanup EXIT

cp scripts/cdp-smoke.js.txt /tmp/cdp-smoke.js
node /tmp/cdp-smoke.js

test -s dist/runtime-smoke.png
SIZE="$(stat -c%s dist/runtime-smoke.png)"
echo "Runtime screenshot: ${SIZE} bytes"

grep -q 'data-starfighter-runtime-initialized="1"' dist/runtime-dom.html
grep -q 'data-starfighter-title-ready="1"' dist/runtime-dom.html
grep -q 'data-starfighter-first-frame="1"' dist/runtime-dom.html
grep -q 'data-starfighter-language-source="ysdk.environment.i18n.lang"' dist/runtime-dom.html
grep -q 'data-starfighter-language="ru"' dist/runtime-dom.html
grep -q 'data-qa-loading-ready="1"' dist/runtime-dom.html
echo "Runtime markers: SDK language ru + Game Ready + title + first-frame OK"

if grep -Eqi 'Aborted\(|RuntimeError:|unhandled exception|failed to asynchronously prepare wasm|wasm streaming compile failed' dist/runtime-browser.log; then
  echo "Runtime abort/error detected in browser log" >&2
  cat dist/runtime-browser.log >&2
  exit 1
fi

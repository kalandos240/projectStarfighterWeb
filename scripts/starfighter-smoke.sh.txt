cat > /tmp/starfighter-csp-server.py <<'PY'
from http.server import ThreadingHTTPServer, SimpleHTTPRequestHandler
from pathlib import Path
import os

CSP = "default-src 'self' data: blob:; script-src 'nonce-starfighter-csp-test' 'self' 'unsafe-eval' 'unsafe-inline'; style-src 'self' 'unsafe-inline'; img-src 'self' data: blob:; media-src 'self' data: blob:; connect-src 'self' data: blob:; worker-src 'self' blob:; font-src 'self' data:;"

class Handler(SimpleHTTPRequestHandler):
    def end_headers(self):
        self.send_header('Content-Security-Policy', CSP)
        self.send_header('Cache-Control', 'no-store')
        super().end_headers()

os.chdir(Path('dist'))
ThreadingHTTPServer(('127.0.0.1', 8765), Handler).serve_forever()
PY

python3 /tmp/starfighter-csp-server.py >dist/runtime-server.log 2>&1 &
SERVER_PID=$!
TARGET_URL="http://127.0.0.1:8765/index.html"
echo "Testing Yandex-mocked CSP launch: $TARGET_URL"

PROFILE_DIR="$(mktemp -d)"
mkdir -p "$PROFILE_DIR/Default"
printf '%s\n' '{"intl":{"accept_languages":"en-US,en"}}' > "$PROFILE_DIR/Default/Preferences"

"$BROWSER" \
  --headless=new \
  --no-sandbox \
  --disable-dev-shm-usage \
  --use-angle=swiftshader \
  --enable-unsafe-swiftshader \
  --autoplay-policy=no-user-gesture-required \
  --enable-logging=stderr \
  --v=0 \
  --lang=en-US \
  --window-size=1280,720 \
  --remote-debugging-address=127.0.0.1 \
  --remote-debugging-port=9222 \
  --user-data-dir="$PROFILE_DIR" \
  "$TARGET_URL" \
  >dist/runtime-browser.stdout 2>dist/runtime-browser.log &
BROWSER_PID=$!

cleanup() {
  kill "$BROWSER_PID" 2>/dev/null || true
  kill "$SERVER_PID" 2>/dev/null || true
  sleep 0.2
  rm -f dist/sdk.js
  rm -rf "$PROFILE_DIR" 2>/dev/null || true
}
trap cleanup EXIT

python3 - <<'PY'

marker = "  console.log('Keyboard input reached the Starfighter main menu');\n"
insert = marker + "\n  const menuShot = await send('Page.captureScreenshot', { format: 'png', captureBeyondViewport: false });\n  fs.writeFileSync('dist/runtime-menu-ru.png', Buffer.from(menuShot.data, 'base64'));\n  console.log('Captured Russian main-menu screenshot under Yandex SDK language mock');\n"
if marker not in text:
    raise SystemExit('main menu screenshot insertion marker not found')

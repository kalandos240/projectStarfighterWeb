from pathlib import Path


def add_emscripten_include(path):
    source = path.read_text()
    old = '#include "SDL.h"\n'
    new = '#include "SDL.h"\n\n#ifdef __EMSCRIPTEN__\n#include <emscripten.h>\n#endif\n'
    if new not in source:
        if old not in source:
            raise SystemExit(f'SDL include marker not found in {path}')
        source = source.replace(old, new, 1)
    path.write_text(source)

from pathlib import Path

path = Path('web/platform-pre.js')
source = path.read_text()

old = """function starfighterReadLocalTimestamp() {
  const metaPath = STARFIGHTER_SAVE_DIR + '/' + STARFIGHTER_META_FILE;
  try {
    if (starfighterFileExists(metaPath)) {
      const meta = JSON.parse(FS.readFile(metaPath, { encoding: 'utf8' }));
      if (Number.isFinite(meta.updatedAt)) return meta.updatedAt;
    }
  } catch (err) {
    console.warn('Could not read Starfighter web metadata:', err);
  }

  let newest = 0;
"""
new = """function starfighterReadLocalTimestamp() {
  const metaPath = STARFIGHTER_SAVE_DIR + '/' + STARFIGHTER_META_FILE;
  let newest = 0;
  try {
    if (starfighterFileExists(metaPath)) {
      const meta = JSON.parse(FS.readFile(metaPath, { encoding: 'utf8' }));
      if (Number.isFinite(meta.updatedAt)) newest = Math.max(newest, meta.updatedAt);
    }
  } catch (err) {
    console.warn('Could not read Starfighter web metadata:', err);
  }

"""
if old not in source:
    raise SystemExit('cloud timestamp marker not found')
source = source.replace(old, new, 1)

old = """      starfighterWriteMetadata(updatedAt);
      document.documentElement.setAttribute('data-starfighter-cloud-saved', '1');
"""
new = """      starfighterWriteMetadata(updatedAt);
      await starfighterPersistLocalStorage();
      document.documentElement.setAttribute('data-starfighter-cloud-saved', '1');
"""
if old not in source:
    raise SystemExit('cloud metadata persistence marker not found')
source = source.replace(old, new, 1)

path.write_text(source)

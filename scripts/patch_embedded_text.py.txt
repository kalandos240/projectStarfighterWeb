from pathlib import Path


def replace_once(path, old, new, label):
    text = path.read_text(encoding='utf-8')
    if new in text:
        return
    if old not in text:
        raise SystemExit(f'{label} marker not found in {path}')
    path.write_text(text.replace(old, new, 1), encoding='utf-8')


title = Path('upstream/src/title.c')
replace_once(title, '\tint skip = 0;\n', '\tint skip = 1;\n', 'skip intro')

replace_once(
    title,
    '''\tSDL_Surface *prlogo, *sflogo;
\tprlogo = gfx_loadImage("gfx/prlogo.png");
\tsflogo = gfx_loadImage("gfx/sflogo.png");
''',
    '''\tSDL_Surface *prlogo;
\tprlogo = gfx_loadImage("gfx/prlogo.png");
''',
    'English title bitmap load',
)

replace_once(

EM_JS(int, starfighter_web_is_russian, (), {
    const sdkLang = globalThis.starfighterYsdk &&
        globalThis.starfighterYsdk.environment &&
        globalThis.starfighterYsdk.environment.i18n &&
        globalThis.starfighterYsdk.environment.i18n.lang;
    const lang = String(sdkLang || navigator.language || '').toLowerCase();
    return lang === 'ru' || lang.startsWith('ru-') || lang.startsWith('ru_');
});

const char *starfighter_web_gettext(const char *source)
{
    if ((source == NULL) || !starfighter_web_is_russian())
        return source;

    for (int i = 0; webTranslations[i].source != NULL; i++)
    {
        if (strcmp(source, webTranslations[i].source) == 0)
            return webTranslations[i].russian;
    }

    return source;
}
